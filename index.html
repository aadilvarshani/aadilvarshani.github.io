title: Office or script host launching a shell with a download cradle or encoded command
tags: [attack.execution, attack.t1059.001, attack.t1204.002, attack.command-and-control, attack.t1105]
logsource: {category: process_creation, product: windows}
detection:
    parent:
        ParentImage|endswith: ['\winword.exe', '\excel.exe', '\powerpnt.exe', '\outlook.exe', '\wscript.exe', '\cscript.exe', '\mshta.exe']
    child:
        Image|endswith: ['\powershell.exe', '\pwsh.exe', '\cmd.exe']
    cradle:
        CommandLine|contains: ['DownloadString', 'DownloadFile', 'Invoke-WebRequest', 'Net.WebClient', 'Start-BitsTransfer']
    encoded:
        CommandLine|contains: [' -enc ', ' /enc ', ' -encodedcommand ']
    hidden:
        CommandLine|contains: [' -w hidden', ' -windowstyle hidden']
    filter_automation:
        ParentCommandLine|contains: '/automation -Embedding'
    condition: parent and child and (cradle or encoded or hidden) and not filter_automation
level: high
(ParentImage IN ("*\\winword.exe", "*\\excel.exe", "*\\powerpnt.exe", "*\\outlook.exe", "*\\wscript.exe", "*\\cscript.exe", "*\\mshta.exe"))
(Image IN ("*\\powershell.exe", "*\\pwsh.exe", "*\\cmd.exe"))
(CommandLine IN ("*DownloadString*", "*DownloadFile*", "*Invoke-WebRequest*", "*Net.WebClient*", "*Start-BitsTransfer*") OR CommandLine IN ("* -enc *", "* /enc *", "* -encodedcommand *") OR CommandLine IN ("* -w hidden*", "* -windowstyle hidden*"))
NOT ParentCommandLine="*/automation -Embedding*"
DeviceProcessEvents
| where (InitiatingProcessFolderPath endswith "\\winword.exe" or InitiatingProcessFolderPath endswith "\\excel.exe" or InitiatingProcessFolderPath endswith "\\powerpnt.exe" or InitiatingProcessFolderPath endswith "\\outlook.exe" or InitiatingProcessFolderPath endswith "\\wscript.exe" or InitiatingProcessFolderPath endswith "\\cscript.exe" or InitiatingProcessFolderPath endswith "\\mshta.exe")
  and (FolderPath endswith "\\powershell.exe" or FolderPath endswith "\\pwsh.exe" or FolderPath endswith "\\cmd.exe")
  and ((ProcessCommandLine contains "DownloadString" or ProcessCommandLine contains "DownloadFile" or ProcessCommandLine contains "Invoke-WebRequest" or ProcessCommandLine contains "Net.WebClient" or ProcessCommandLine contains "Start-BitsTransfer") or (ProcessCommandLine contains " -enc " or ProcessCommandLine contains " /enc " or ProcessCommandLine contains " -encodedcommand ") or (ProcessCommandLine contains " -w hidden" or ProcessCommandLine contains " -windowstyle hidden"))
  and (not(InitiatingProcessCommandLine contains "/automation -Embedding"))
any where (process.parent.executable.caseless like~ ("*\\winword.exe", "*\\excel.exe", "*\\powerpnt.exe", "*\\outlook.exe", "*\\wscript.exe", "*\\cscript.exe", "*\\mshta.exe"))
  and (process.executable.caseless like~ ("*\\powershell.exe", "*\\pwsh.exe", "*\\cmd.exe"))
  and ((process.command_line like~ ("*DownloadString*", "*DownloadFile*", "*Invoke-WebRequest*", "*Net.WebClient*", "*Start-BitsTransfer*")) or (process.command_line like~ ("* -enc *", "* /enc *", "* -encodedcommand *")) or (process.command_line like~ ("* -w hidden*", "* -windowstyle hidden*")))
  and (not process.parent.command_line:"*/automation -Embedding*")
rule office_or_script_host_launching_a_shell_with_a_download_cradle_or_encoded_command {
  meta:
    id = "6b1f0c2e-8d4a-4e57-9b3c-2f7a1d9e0c41"
    title = "Office or script host launching a shell with a download cradle or encoded command"
    description = "An Office application or Windows script host starts PowerShell or cmd with a download cradle, an encoded command or a hidden window. Office COM automation is filtered out."
    author = "Aadil Varshani"
    reference = ""
    date = "2026-11-06"
    tags = "attack.execution, attack.t1059.001, attack.t1204.002, attack.command-and-control, attack.t1105"
    severity = "high"
    falsepositives = "Admin scripts launched from Office macros in managed environments"

  events:
    $event1.metadata.event_type = "PROCESS_LAUNCH"
    $event1.principal.process.file.full_path = /\\winword\.exe|\\excel\.exe|\\powerpnt\.exe|\\outlook\.exe|\\wscript\.exe|\\cscript\.exe|\\mshta\.exe/ nocase
      AND $event1.target.process.file.full_path = /\\powershell\.exe|\\pwsh\.exe|\\cmd\.exe/ nocase AND ($event1.target.process.command_line = /DownloadString|DownloadFile|Invoke-WebRequest|Net\.WebClient|Start-BitsTransfer/ nocase OR $event1.target.process.command_line = / -enc | \/enc | -encodedcommand / nocase OR $event1.target.process.command_line = / -w hidden| -windowstyle hidden/ nocase)
      AND $event1.principal.process.command_line != /\/automation -Embedding/ nocase

  conditions:
    $event1
}

Building the sensor

ReaperWatch, open source

Alongside the day job, I built the last seat myself. ReaperWatch is a Windows EDR: a C++ agent on ETW that enriches every process launch, and a Node.js console that runs it through an EQL rule engine. Four releases in, kernel driver next.

  1. ETWkernel logger, real time
  2. Enrichlineage, hashes, signer
  3. JSONLdurable, on disk
  4. Ingesttail into SQLite
  5. EQLevery event, every rule
  6. Alertseverity, ATT&CK
